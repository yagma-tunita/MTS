第五步：创建订单
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

$body = @{shipper_company_id=1;city_id=1;line_id=1;vessel_id=1;voyage_date="2026-07-15";start_port_id=1;end_port_id=3;cargo_items=@(@{cargo_name="钢材";cargo_type="bulk";quantity=100;weight_ton=500;volume_cub_m=200;unit_price=50;subtotal=5000});shipper_contact="张三-13800138000";consignee_contact="李四-13900139000"} | ConvertTo-Json -Depth 10

$orderResp = Invoke-RestMethod -Uri "$base/orders" -Method Post -Body $body -ContentType "application/json" -Headers $h
$orderResp | ConvertTo-Json -Depth 10
$oid = $orderResp.data.order_id
Write-Host "订单创建成功 ID=$oid" -ForegroundColor Green

⚠ 常见错误：
  · {"code":1002,"message":"shipper_company_id mismatch"}
    原因：JWT 中的 user_id 不等于请求中的 shipper_company_id
    解决：用 $login.data.user_id 的值作为 shipper_company_id

  · {"code":1003,"message":"no LOAD cargo note for start port"}
    原因：该航线/船舶/日期下，出发港没有 LOAD 类型的舱单记录
    解决：检查 seed 数据是否正确，或先调用航次推荐确认可用航次

  · {"code":1003,"message":"no UNLOAD cargo note for end port"}
    同上，目的港没有 UNLOAD 类型的舱单记录

  · {"code":1003,"message":"vessel not found or max deadweight missing"}
    原因：vessel_id 不存在或船舶没有设置最大载重吨
    解决：先查询船舶列表确认 vessel_id

  · {"code":1003,"message":"shipping line not found or port sequence missing"}
    原因：line_id 不存在或航线缺少港口序列
    解决：先查询航线列表确认 line_id

  · {"code":1004,"message":"insufficient capacity, remaining: xxx"}
    原因：所选航段剩余运力不足
    解决：减少 cargo_items 中的 weight_ton，或换个航次

  · {"code":1005,"message":"failed to acquire lock, please retry"}
    原因：并发锁冲突，同一航线/船舶/日期同时有多个订单在创建
    解决：重试即可

  · {"code":2000,"message":"failed to create order"}
    原因：服务器内部错误，看服务端窗口的详细日志
    常见子原因：
      - SegmentCapacityUsage 模型与数据库表不匹配（delete_time 列）
      - 外键约束失败（如 city_id 不存在）
      - GORM 插入错误

  · cargo_items 不能为空数组，至少 1 项
  · voyage_date 必须格式为 YYYY-MM-DD

# 注意：订单创建后 order_status = 1（已确认），所以从 1→2 开始
Write-Host "=== 开始运输 (1→2) ===" -ForegroundColor Cyan
Invoke-RestMethod -Uri "$base/orders/$oid/status" -Method Put -Body '{"status":2}' -ContentType "application/json" -Headers $h

Write-Host "=== 完成运输 (2→3) ===" -ForegroundColor Cyan
Invoke-RestMethod -Uri "$base/orders/$oid/status" -Method Put -Body '{"status":3}' -ContentType "application/json" -Headers $h

⚠ 状态转换规则（禁止跳转和回退）：
  0(草稿) → 1(已确认) 或 4(取消)
  1(已确认) → 2(运输中) 或 4(取消)
  2(运输中) → 3(已完成) 或 4(取消)
  3(已完成) → 不可再变（终态）
  4(已取消) → 不可再变（终态）

⚠ 常见错误：
  · {"code":1000,"message":"invalid state transition"}
    原因：非法状态转换，例如 1→1（同状态）、2→1（回退）、3→其他（终态）
    解决：先 GET 查当前 order_status，再走合法路径

# 在浏览器 F12 控制台执行（先用 test001 登录获取 token）：
var token = "把第一步打印的完整 access_token 粘贴到这里";
var ws = new WebSocket("ws://localhost:8080/ws?token=" + token);
ws.onopen = function() { console.log("已连接"); };
ws.onmessage = function(e) { console.log("收到:", JSON.parse(e.data)); };
ws.onerror = function(e) { console.error("错误:", e); };
ws.onclose = function() { console.log("已断开"); };

# 然后在另一个窗口更新订单状态，WebSocket 会收到推送消息

⚠ 常见错误：
  · 连接立即断开 → token 过期或无效，重新登录获取新 token
  · 连接被拒绝 401 → WebSocket 路径是 /ws 不是在 /api/v1 下
  · 收不到推送 → 更新订单状态时，推送目标 user_id 必须与 WebSocket 登录的一致
  · 生产环境 WebSocket 需要独立的负载均衡策略

Write-Host "=== 注册新货主 ===" -ForegroundColor Cyan
Invoke-RestMethod -Uri "$base/shipper/register" -Method Post -Body '{"company_name":"新货主","login_username":"newshipper","password":"123456"}' -ContentType "application/json"

Write-Host "=== 新货主登录 ===" -ForegroundColor Cyan
$newLogin = Invoke-RestMethod -Uri "$base/auth/login" -Method Post -Body '{"username":"newshipper","password":"123456","role":"shipper"}' -ContentType "application/json"
Write-Host "新货主 user_id=$($newLogin.data.user_id)" -ForegroundColor Green

Write-Host "=== 新货主改密码 ===" -ForegroundColor Cyan
$newH = @{Authorization="Bearer $($newLogin.data.access_token)"}
Invoke-RestMethod -Uri "$base/shipper/password/$($newLogin.data.user_id)" -Method Post -Body '{"old_password":"123456","new_password":"abcdef"}' -ContentType "application/json" -Headers $newH

⚠ 常见错误：
  · {"code":2000,"message":"failed to register"}
    原因：用户名已被注册（唯一索引冲突），或密码强度不足（需 ≥6 位）
    解决：换一个 login_username
  · {"code":1000,"message":"Key: 'registerShipperRequest.Password' Error:...len=3"}
    原因：密码太短，至少 6 位
  · 改密码时 :id 必须等于当前登录用户的 user_id，否则 403

# 前置条件：config.yaml 已配置 notify.email（smtp.qq.com:465 + 授权码）
# 目标邮箱填你想收到邮件的地址，可以填自己的 QQ 邮箱

$adminLogin = Invoke-RestMethod -Uri "$base/auth/login" -Method Post -Body '{"username":"admin","password":"admin123","role":"admin"}' -ContentType "application/json"
$ah = @{Authorization="Bearer $($adminLogin.data.access_token)"}

$body = @{
  user_id = 1
  role    = "shipper"
  type    = "order_created"
  title   = "MTS系统测试邮件"
  content = "这是一封来自MTS系统的测试邮件，收到说明SMTP配置正确。"
    data    = @{ email = "your_qq_number@qq.com" }
} | ConvertTo-Json -Depth 10

Invoke-RestMethod -Uri "$base/admin/notifications" -Method Post -Body $body -ContentType "application/json" -Headers $ah

⚠ 常见错误：
  · 邮件没收到 → 检查服务端窗口是否打印 "email send failed" 及具体错误
  · 没有报错但邮件没到 → 可能在垃圾箱，或 QQ 邮箱的 SMTP 频率限制
  · 授权码错误 → 去 QQ邮箱→设置→账户→重新生成授权码
  · from_name 含中文可能被拒收，保持 "MTS System" 即可

附：常见 HTTP 状态码速查
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

200 → 成功（业务 code=0）
400 → 请求参数错误（业务 code=1000）
401 → 未授权（业务 code=1001），token 缺失/过期/无效
403 → 权限不足（业务 code=1002），角色不匹配或 user_id 不匹配
404 → 资源未找到（业务 code=1003）
409 → 冲突（业务 code=1004），运力不足/重复取消
429 → 限流（业务 code=1005），请求太频繁
500 → 服务器内部错误（业务 code=2000），看服务端日志
